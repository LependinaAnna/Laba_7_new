int z_1(char c)
{
    int res_1;
    res_1 = (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z');
    return res_1;
}

float z_2(float x, char C, float y)
{
    float res_2;
    if (C == '+')
    {
        res_2 = x + y;
    }
    else if (C == '-')
    {
        res_2 = x - y;
    }
    else if (C == '*')
    {
        res_2 = x * y;
    }
    else if (C == '/')
    {
        if (y != 0.0f) 
        {
            res_2 = x / y;
        }
    }
    else if (C == '^')
    {
        res_2 = pow(x, y);
    }
    return res_2;
}
